<a id="l00159"></a>/*
<a id="l00160"></a> * The GCC builtins __builtin_clz(), __builtin_ctz(), and 64-bit
<a id="l00161"></a> * variants are described by the GCC documentation as having undefined
<a id="l00162"></a> * behavior when the argument is zero. See
<a id="l00163"></a> * https://gcc.gnu.org/onlinedocs/gcc/Other-Builtins.html.
<a id="l00164"></a> *
<a id="l00165"></a> * The undefined behavior applies to all architectures, regardless of
<a id="l00166"></a> * the behavior of the instruction used to implement the builtin.
<a id="l00167"></a> *
<a id="l00168"></a> * We don't want to expose users of this API to the undefined behavior,
<a id="l00169"></a> * so we use a conditional to explicitly provide the correct result when
<a id="l00170"></a> * x=0.
<a id="l00171"></a> *
<a id="l00172"></a> * Most instruction set architectures have a CLZ instruction or similar
<a id="l00173"></a> * that already computes the correct result for x=0. Both GCC and Clang
<a id="l00174"></a> * know this and simply generate a CLZ instruction, optimizing away the
<a id="l00175"></a> * conditional.
<a id="l00176"></a> *
<a id="l00177"></a> * For x86, and for compilers that fail to eliminate the conditional,
<a id="l00178"></a> * there is often another opportunity for optimization since code using
<a id="l00179"></a> * these functions tends to contain a zero check already. For example,
<a id="l00180"></a> * from kernel/sched.c:
<a id="l00181"></a> *
<a id="l00182"></a> *      struct k_thread *z_priq_mq_best(struct _priq_mq *pq)
<a id="l00183"></a> *      {
<a id="l00184"></a> *              if (!pq-&gt;bitmask) {
<a id="l00185"></a> *                      return NULL;
<a id="l00186"></a> *              }
<a id="l00187"></a> *
<a id="l00188"></a> *              struct k_thread *thread = NULL;
<a id="l00189"></a> *              sys_dlist_t *l =
<a id="l00190"></a> *                      &amp;pq-&gt;queues[u32_count_trailing_zeros(pq-&gt;bitmask)];
<a id="l00191"></a> *
<a id="l00192"></a> *              ...
<a id="l00193"></a> *
<a id="l00194"></a> * The compiler will often be able to eliminate the redundant x == 0
<a id="l00195"></a> * check after inlining the call to u32_count_trailing_zeros().
<a id="l00196"></a> */
<a id="l00197"></a>
<a id="l00198"></a>#if use_builtin(__builtin_clz)
<a id="l00199"></a>static inline int u32_count_leading_zeros(uint32_t x)
<a id="l00200"></a>{
<a id="l00201"></a>        return (x == 0) ? 32 : __builtin_clz(x);
<a id="l00202"></a>}
<a id="l00203"></a>#else /* !use_builtin(__builtin_clz) */
<a id="l00204"></a>static inline int u32_count_leading_zeros(uint32_t x)
<a id="l00205"></a>{
<a id="l00206"></a>        int b;
<a id="l00207"></a>
<a id="l00208"></a>        for (b = 0; b &lt; 32 &amp;&amp; (x &gt;&gt; 31) == 0; b++) {
<a id="l00209"></a>                x &lt;&lt;= 1;
<a id="l00210"></a>        }
<a id="l00211"></a>
<a id="l00212"></a>        return b;
<a id="l00213"></a>}
<a id="l00214"></a>#endif /* use_builtin(__builtin_clz) */
<a id="l00215"></a>
<a id="l00216"></a>#if use_builtin(__builtin_clzll)
<a id="l00217"></a>static inline int u64_count_leading_zeros(uint64_t x)
<a id="l00218"></a>{
<a id="l00219"></a>        return (x == 0) ? 64 : __builtin_clzll(x);
<a id="l00220"></a>}
<a id="l00221"></a>#else /* !use_builtin(__builtin_clzll) */
<a id="l00222"></a>static inline int u64_count_leading_zeros(uint64_t x)
<a id="l00223"></a>{
<a id="l00224"></a>        if (x == (uint32_t)x) {
<a id="l00225"></a>                return 32 + u32_count_leading_zeros((uint32_t)x);
<a id="l00226"></a>        } else {
<a id="l00227"></a>                return u32_count_leading_zeros(x &gt;&gt; 32);
<a id="l00228"></a>        }
<a id="l00229"></a>}
<a id="l00230"></a>#endif /* use_builtin(__builtin_clzll) */

<a id="l00264"></a>#endif /* use_builtin(__builtin_ctzll) */
<a id="l00265"></a>
<a id="l00276"></a>#if __has_type_128
<a id="l00277"></a>static inline void i128_multiply_i64_i64(int64_t a, int64_t b, int128_t *result)
<a id="l00278"></a>{
<a id="l00279"></a>        __int128 c = (__int128)a * (__int128)b;
<a id="l00280"></a>
<a id="l00281"></a>        result-&gt;low = (uint64_t)c;
<a id="l00282"></a>        result-&gt;high = (uint64_t)(c &gt;&gt; 64);
<a id="l00283"></a>}
<a id="l00284"></a>#else
<a id="l00285"></a>static inline void i128_multiply_i64_i64(int64_t a, int64_t b, int128_t *result)
<a id="l00286"></a>{
<a id="l00287"></a>        uint64_t u_a = (a &lt; 0) ? (uint64_t)-a : (uint64_t)a;
<a id="l00288"></a>        uint64_t u_b = (b &lt; 0) ? (uint64_t)-b : (uint64_t)b;
<a id="l00289"></a>        int sign = (a &lt; 0) ^ (b &lt; 0);
<a id="l00290"></a>
<a id="l00291"></a>        /* Split to 32-bit values */
<a id="l00292"></a>        uint64_t a_lo = u_a &amp; 0xFFFFFFFFULL;
<a id="l00293"></a>        uint64_t a_hi = u_a &gt;&gt; 32;
<a id="l00294"></a>        uint64_t b_lo = u_b &amp; 0xFFFFFFFFULL;
<a id="l00295"></a>        uint64_t b_hi = u_b &gt;&gt; 32;
<a id="l00296"></a>
<a id="l00297"></a>        /* Calculate product, just like in school */
<a id="l00298"></a>        uint64_t res_0 = a_lo * b_lo;
<a id="l00299"></a>        uint64_t res_1 = a_hi * b_lo;
<a id="l00300"></a>        uint64_t res_2 = a_lo * b_hi;
<a id="l00301"></a>        uint64_t res_3 = a_hi * b_hi;
<a id="l00302"></a>
<a id="l00303"></a>        /* Combine values including carry */
<a id="l00304"></a>        uint64_t carry = 0;
<a id="l00305"></a>        uint64_t middle = (res_0 &gt;&gt; 32) + (res_1 &amp; 0xFFFFFFFFULL) + (res_2 &amp; 0xFFFFFFFFULL);
<a id="l00306"></a>
<a id="l00307"></a>        result-&gt;low = (res_0 &amp; 0xFFFFFFFFULL) | (middle &lt;&lt; 32);
<a id="l00308"></a>
<a id="l00309"></a>        /* Move the top part */
<a id="l00310"></a>        carry = (middle &gt;&gt; 32) + (res_1 &gt;&gt; 32) + (res_2 &gt;&gt; 32) + res_3;
<a id="l00311"></a>        result-&gt;high = carry;
<a id="l00312"></a>
<a id="l00313"></a>        /* Calculate two-complement if sign is minus */
<a id="l00314"></a>        if (sign) {
<a id="l00315"></a>                result-&gt;low = ~result-&gt;low + 1;
<a id="l00316"></a>                result-&gt;high = ~result-&gt;high + (result-&gt;low == 0 ? 1 : 0);
<a id="l00317"></a>        }
<a id="l00318"></a>}
<a id="l00319"></a>#endif /* __has_type_128 */

<a id="l00322"></a>
<a id="l00324"></a>

## 5.4\_源码跟踪译注

> 译注——返回方向：加法与乘法接口返回true表示发生溢出，false才表示结果可由目标类型完整表示。result仍必须指向有效、可写且类型匹配的对象；错误标志不是“完全没有写入”的保证，也不是饱和运算（Saturating Arithmetic）。

> 译注——类型与输入：u16、u32、u64和size接口分别按其参数类型进行计算。较宽或有符号的表达式传入时可能先发生类型转换；函数只能检查转换后的操作数，不能追溯传参前已丢失的数据。size_t的位宽取决于目标ABI。

> 译注——选择分支：实现按HAS_BUILTIN选择相应编译器内建函数；128位乘法还检查__SIZEOF_INT128__。在首次包含公共头文件前定义PORTABLE_MISC_MATH_EXTRAS可强制后备分支，主要供测试使用；源码注释中的<misc/math_extras.h>是历史路径，当前公共入口是<zephyr/sys/math_extras.h>。

> 译注——后备分支写作uint16_t c = a * b。在32位int能够表示全部uint16_t取值的平台上，两个实参先提升为int再相乘；例如65535*65535超过INT_MAX，可能发生有符号溢出的未定义行为（Undefined Behavior），不能认为赋值到uint16_t会消除此前的问题。内建函数分支不使用这个乘法表达式；这里保留原源码并指出分支差异，没有修改上游实现或宣称实测通过。

> 译注——原文“无溢出”描述的是两个有符号64位数相乘可由128位容纳的数学范围。当前无原生128位类型的后备实现先计算(uint64_t)-a及(uint64_t)-b；当操作数为INT64_MIN时，有符号取负发生在无符号转换之前，结果不可由int64_t表示，存在未定义行为。不能把数学范围保证直接推广为该后备代码对全部输入均可靠；原生__int128分支不经过这条取绝对值路径。

> 译注——返回的是零位数量，不是置位数量，也不是固定意义上的位索引。x=0时，32位版本返回32、64位版本返回64；将结果作为数组索引或移位量前仍需检查零输入。此封装显式处理0，不能用不处理0的__builtin_clz／__builtin_ctz调用直接替换。
