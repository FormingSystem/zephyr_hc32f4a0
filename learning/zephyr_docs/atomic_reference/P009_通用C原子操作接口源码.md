```c
/*
 * Copyright (c) 1997-2015, Wind River Systems, Inc.
 *
 * SPDX-License-Identifier: Apache-2.0
 */


/* IWYU pragma: private, include "zephyr/sys/atomic.h" */

#ifndef ZEPHYR_INCLUDE_SYS_ATOMIC_C_H_
#define ZEPHYR_INCLUDE_SYS_ATOMIC_C_H_

/* Included from <atomic.h> */

#ifdef __cplusplus
extern "C" {
#endif


/* Simple and correct (but very slow) implementation of atomic
 * primitives that require nothing more than kernel interrupt locking.
 */

__syscall bool atomic_cas(atomic_t *target, atomic_val_t old_value,
                         atomic_val_t new_value);

__syscall bool atomic_ptr_cas(atomic_ptr_t *target, atomic_ptr_val_t old_value,
                              atomic_ptr_val_t new_value);

__syscall atomic_val_t atomic_add(atomic_t *target, atomic_val_t value);

__syscall atomic_val_t atomic_sub(atomic_t *target, atomic_val_t value);

static inline atomic_val_t atomic_inc(atomic_t *target)
{
        return atomic_add(target, 1);

}

static inline atomic_val_t atomic_dec(atomic_t *target)
{
        return atomic_sub(target, 1);

}

atomic_val_t atomic_get(const atomic_t *target);

atomic_ptr_val_t atomic_ptr_get(const atomic_ptr_t *target);

__syscall atomic_val_t atomic_set(atomic_t *target, atomic_val_t value);

__syscall atomic_ptr_val_t atomic_ptr_set(atomic_ptr_t *target, atomic_ptr_val_t value);

static inline atomic_val_t atomic_clear(atomic_t *target)
{
        return atomic_set(target, 0);

}

static inline atomic_ptr_val_t atomic_ptr_clear(atomic_ptr_t *target)
{
        return atomic_ptr_set(target, NULL);

}

__syscall atomic_val_t atomic_or(atomic_t *target, atomic_val_t value);

__syscall atomic_val_t atomic_xor(atomic_t *target, atomic_val_t value);

__syscall atomic_val_t atomic_and(atomic_t *target, atomic_val_t value);

__syscall atomic_val_t atomic_nand(atomic_t *target, atomic_val_t value);


#ifdef __cplusplus
}
#endif

#ifdef CONFIG_ATOMIC_OPERATIONS_C

#ifndef DISABLE_SYSCALL_TRACING
/* Skip defining macros of atomic_*() for syscall tracing.
 * Compiler does not like "({ ... tracing code ... })" and complains
 *
 *   error: expected identifier or '(' before '{' token
 *
 * ... even though there is a '(' before '{'.
 */
#define DISABLE_SYSCALL_TRACING
#define _REMOVE_DISABLE_SYSCALL_TRACING
#endif

#include <zephyr/syscalls/atomic_c.h>

#ifdef _REMOVE_DISABLE_SYSCALL_TRACING
#undef DISABLE_SYSCALL_TRACING
#undef _REMOVE_DISABLE_SYSCALL_TRACING
#endif

#endif

#endif /* ZEPHYR_INCLUDE_SYS_ATOMIC_C_H_ */
```

## 9.2\_可见注释译文（Comments）

1. 版权归1997–2015年的Wind River Systems所有；许可证为Apache-2.0。

2. 包含项检查工具Include What You Use（IWYU）的私有头文件标记：使用方应包含公共头文件zephyr/sys/atomic.h。

3. 由atomic.h包含。

4. 原子原语的一种简单而正确、但很慢的实现，只需内核中断锁定机制。

5. 跳过为系统调用跟踪（Syscall Tracing）定义atomic_*()宏。编译器不接受“({ ... tracing code ... })”，并报告“在{记号之前需要标识符或(”的错误，尽管{前面实际上已经有(。原始诊断保留在代码注释中。

6. ZEPHYR_INCLUDE_SYS_ATOMIC_C_H_头文件保护条件结束。

## 9.3\_源码跟踪译注

> 译注——声明与实现：本头文件声明需要系统调用（System Call）的操作，并把inc／dec及clear包装为add／sub或set。真正的C实现位于lib/os/atomic_c.c；带__syscall的声明不是这里就完成了读改写。

> 译注——SMP边界：本次atomic_c.c的读改写使用单个全局k_spinlock；z_impl_atomic_cas中有BUILD_ASSERT(!IS_ENABLED(CONFIG_SMP))，因为SMP下k_spin_lock又调用atomic_cas，会发生递归。不能仅凭文件开头提到其他处理器，就声称此C后端支持对称多处理（Symmetric Multiprocessing，SMP）；应选适配架构的ARCH或BUILTIN实现。

> 译注——读取边界：本次C实现的atomic_get和atomic_ptr_get直接return *target，没有像内建后端那样显式调用__atomic_load_n。其旧注释提到4字节对齐；该措辞不能替代目标ABI的对齐要求。公共接口要求与特定后端实现应分别核对，不能把普通C指针读取推广为任意多核平台上的同步方式。

> 译注——跟踪宏的作用域：仅在CONFIG_ATOMIC_OPERATIONS_C下包含生成的syscalls/atomic_c.h。如果原先未定义DISABLE_SYSCALL_TRACING，则临时定义并设置移除标志，包含后恢复；已有的定义不会被这里取消。

实现定位：[本地C源码](../../../lib/os/atomic_c.c)与[官方C源码](https://github.com/zephyrproject-rtos/zephyr/blob/main/lib/os/atomic_c.c)；配置依据见[官方Kconfig.atomics](https://github.com/zephyrproject-rtos/zephyr/blob/main/lib/os/Kconfig.atomics)。

仓库相对定位：[include/zephyr/sys/atomic_c.h](../../../include/zephyr/sys/atomic_c.h)；本次核对的[官方原始头文件](https://github.com/zephyrproject-rtos/zephyr/blob/main/include/zephyr/sys/atomic_c.h)摘要见[来源记录](../SOURCES.md)。去掉注释和展示空白后，本页代码与本次读取的原始头文件一致；不能把这种一致性推广到以后滚动更新的main。
