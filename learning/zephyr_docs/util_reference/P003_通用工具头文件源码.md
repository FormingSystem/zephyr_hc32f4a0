<a id="l00617"></a>static inline int64_t arithmetic_shift_right(int64_t value, uint8_t shift)
<a id="l00618"></a>{
<a id="l00619"></a>        int64_t sign_ext;
<a id="l00620"></a>
<a id="l00621"></a>        if (shift == 0U) {
<a id="l00622"></a>                return value;
<a id="l00623"></a>        }
<a id="l00624"></a>
<a id="l00625"></a>        /* extract sign bit */
<a id="l00626"></a>        sign_ext = (value &gt;&gt; 63) &amp; 1;
<a id="l00627"></a>
<a id="l00628"></a>        /* make all bits of sign_ext be the same as the value's sign bit */
<a id="l00629"></a>        sign_ext = -sign_ext;
<a id="l00630"></a>
<a id="l00631"></a>        /* shift value and fill opened bit positions with sign bit */
<a id="l00632"></a>        return (value &gt;&gt; shift) | (sign_ext &lt;&lt; (64 - shift));
<a id="l00633"></a>}
