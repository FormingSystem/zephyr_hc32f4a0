```c
/*
 * Copyright (c) 2015, Wind River Systems, Inc.
 * Copyright (c) 2017, Oticon A/S
 *
 * SPDX-License-Identifier: Apache-2.0

/* Memory mapped registers I/O functions in non-arch-specific C code */

#ifndef ZEPHYR_INCLUDE_ARCH_COMMON_SYS_IO_H_
#define ZEPHYR_INCLUDE_ARCH_COMMON_SYS_IO_H_

#ifndef _ASMLANGUAGE

#include <zephyr/toolchain.h>
#include <zephyr/types.h>
#include <zephyr/sys/sys_io.h>

#ifdef __cplusplus
extern "C" {
#endif


static ALWAYS_INLINE uint8_t sys_read8(mem_addr_t addr)
{
        return *(volatile uint8_t *)addr;
}

static ALWAYS_INLINE void sys_write8(uint8_t data, mem_addr_t addr)
{
        *(volatile uint8_t *)addr = data;
}

static ALWAYS_INLINE uint16_t sys_read16(mem_addr_t addr)
{
        return *(volatile uint16_t *)addr;
}

static ALWAYS_INLINE void sys_write16(uint16_t data, mem_addr_t addr)
{
        *(volatile uint16_t *)addr = data;
}

static ALWAYS_INLINE uint32_t sys_read32(mem_addr_t addr)
{
        return *(volatile uint32_t *)addr;
}

static ALWAYS_INLINE void sys_write32(uint32_t data, mem_addr_t addr)
{
        *(volatile uint32_t *)addr = data;
}

static ALWAYS_INLINE uint64_t sys_read64(mem_addr_t addr)
{
        return *(volatile uint64_t *)addr;
}

static ALWAYS_INLINE void sys_write64(uint64_t data, mem_addr_t addr)
{
        *(volatile uint64_t *)addr = data;
}


#ifdef __cplusplus
}
#endif

#endif /* _ASMLANGUAGE */

#endif /* ZEPHYR_INCLUDE_ARCH_COMMON_SYS_IO_H_ */
```

## 24.3\_源码跟踪译注

> 译注——访问宽度：八个函数分别对volatile uint8_t、uint16_t、uint32_t和uint64_t指针直接读写，写操作参数顺序为data、addr。这些函数内没有地址映射、端序转换、显式屏障或锁；地址有效性、对齐和外设访问协议仍须满足。64位C表达式也不能据此推断为单次原子总线事务。

> 译注——声明类型：公共文档对寄存器地址使用mm_reg_t，本通用实现形参使用mem_addr_t；两者在公共头文件中都定义为uintptr_t。这是当前类型别名的选择，不表示函数会把普通内存地址自动转换为MMIO映射。

仓库相对定位：[include/zephyr/arch/common/sys_io.h](../../../include/zephyr/arch/common/sys_io.h)；本次核对的[官方原始头文件](https://github.com/zephyrproject-rtos/zephyr/blob/main/include/zephyr/arch/common/sys_io.h)摘要见[来源记录](../SOURCES.md)。去掉注释和展示空白后，本页代码与本次读取的原始头文件一致；不能把这种一致性推广到以后滚动更新的main。
