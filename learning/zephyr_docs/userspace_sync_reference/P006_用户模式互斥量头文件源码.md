```c
/*
 * Copyright (c) 2019 Intel Corporation
 *
 * SPDX-License-Identifier: Apache-2.0
 */

#ifndef ZEPHYR_INCLUDE_SYS_MUTEX_H_
#define ZEPHYR_INCLUDE_SYS_MUTEX_H_

/*
 * sys_mutex behaves almost exactly like k_mutex, with the added advantage
 * that a sys_mutex instance can reside in user memory.
 *
 * Further enhancements will support locking/unlocking uncontended sys_mutexes
 * with simple atomic ops instead of syscalls, similar to Linux's
 * FUTEX_LOCK_PI and FUTEX_UNLOCK_PI
 */

#ifdef __cplusplus
extern "C" {
#endif

#ifdef CONFIG_USERSPACE
#include <zephyr/sys/atomic.h>
#include <zephyr/types.h>
#include <zephyr/sys/clock.h>

struct sys_mutex {
        /* Currently unused, but will be used to store state for fast mutexes
         * that can be locked/unlocked with atomic ops if there is no
         * contention
         */
        atomic_t val;
};


#define SYS_MUTEX_DEFINE(name) \
        struct sys_mutex name

static inline void sys_mutex_init(struct sys_mutex *mutex)
{
        ARG_UNUSED(mutex);

        /* Nothing to do, kernel-side data structures are initialized at
         * boot
         */
}

__syscall int z_sys_mutex_kernel_lock(struct sys_mutex *mutex,
                                      k_timeout_t timeout);

__syscall int z_sys_mutex_kernel_unlock(struct sys_mutex *mutex);

static inline int sys_mutex_lock(struct sys_mutex *mutex, k_timeout_t timeout)
{
        /* For now, make the syscall unconditionally */
        return z_sys_mutex_kernel_lock(mutex, timeout);
}

static inline int sys_mutex_unlock(struct sys_mutex *mutex)
{
        /* For now, make the syscall unconditionally */
        return z_sys_mutex_kernel_unlock(mutex);
}

#include <zephyr/syscalls/mutex.h>

#else
#include <zephyr/kernel.h>

struct sys_mutex {
        struct k_mutex kernel_mutex;
};

#define SYS_MUTEX_DEFINE(name) \
        struct sys_mutex name = { \
                .kernel_mutex = Z_MUTEX_INITIALIZER(name.kernel_mutex) \
        }

static inline void sys_mutex_init(struct sys_mutex *mutex)
{
        k_mutex_init(&mutex->kernel_mutex);
}

static inline int sys_mutex_lock(struct sys_mutex *mutex, k_timeout_t timeout)
{
        return k_mutex_lock(&mutex->kernel_mutex, timeout);
}

static inline int sys_mutex_unlock(struct sys_mutex *mutex)
{
        return k_mutex_unlock(&mutex->kernel_mutex);
}

#endif /* CONFIG_USERSPACE */


#ifdef __cplusplus
}
#endif

#endif /* ZEPHYR_INCLUDE_SYS_MUTEX_H_ */
```

## 6.2\_可见注释译文（Comments）

1. 版权归2019年的Intel Corporation所有；Apache-2.0许可证标识与原署名保持原样。

2. sys_mutex的行为几乎与k_mutex相同，额外优势是实例可位于用户内存。后续改进计划在无竞争（Uncontended）时用简单原子操作完成加锁／解锁，替代系统调用（System Call），类似Linux的FUTEX_LOCK_PI和FUTEX_UNLOCK_PI。此段描述未来增强，并非已实现的快速路径。

3. 当前未使用；计划用于保存快速互斥量（Fast Mutex）的状态，以便无竞争时通过原子操作加锁／解锁。

4. 无需操作，内核侧数据结构在启动时初始化。

5. 目前无条件调用系统调用接口。

6. 目前无条件调用系统调用接口。

7. CONFIG_USERSPACE条件分支结束标记。

8. ZEPHYR_INCLUDE_SYS_MUTEX_H_头文件保护条件结束。

## 6.3\_源码跟踪译注

> 译注——配置决定布局：CONFIG_USERSPACE下sys_mutex只有atomic_t val；未启用时包含struct k_mutex kernel_mutex。原子字段当前未参与快速加锁，不能自行把val写成0／1当作控制此互斥量。

> 译注——内核关联：用户模式分支的lock／unlock总调用z_sys_mutex_kernel_*。kernel/userspace/mutex.c以地址查询K_OBJ_SYS_MUTEX，再取obj->data.mutex；用户态入口另检查对象地址的可写访问权限。地址可写并不自动意味着对象已被内核识别。

> 译注——初始化边界：用户模式的sys_mutex_init不执行实际初始化，注释说明底层对象在启动时准备；未启用用户模式时才调用嵌入对象的k_mutex_init。头文件关于无竞争原子操作的说明是未来改进，不应写成当前已存在的性能保证。

实现定位：[仓库kernel/userspace/mutex.c](../../../kernel/userspace/mutex.c)与[官方实现](https://github.com/zephyrproject-rtos/zephyr/blob/main/kernel/userspace/mutex.c)。

仓库相对定位：[include/zephyr/sys/mutex.h](../../../include/zephyr/sys/mutex.h)；本次核对的[官方原始头文件](https://github.com/zephyrproject-rtos/zephyr/blob/main/include/zephyr/sys/mutex.h)摘要见[来源记录](../SOURCES.md)。去掉注释和展示空白后，本页代码与本次读取的原始头文件一致；不能把这种一致性推广到以后滚动更新的main。
