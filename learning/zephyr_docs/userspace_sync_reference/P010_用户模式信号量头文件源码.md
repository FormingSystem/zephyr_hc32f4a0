```c
/*
 * Copyright (c) 2019 Intel Corporation
 *
 * SPDX-License-Identifier: Apache-2.0
 */


#ifndef ZEPHYR_INCLUDE_SYS_SEM_H_
#define ZEPHYR_INCLUDE_SYS_SEM_H_

/*
 * sys_sem exists in user memory working as counter semaphore for
 * user mode thread when user mode enabled. When user mode isn't
 * enabled, sys_sem behaves like k_sem.
 */

#include <zephyr/kernel.h>
#include <zephyr/sys/atomic.h>
#include <zephyr/types.h>
#include <zephyr/sys/iterable_sections.h>
#include <zephyr/sys/__assert.h>

#ifdef __cplusplus
extern "C" {
#endif

struct sys_sem {
#ifdef CONFIG_USERSPACE
        struct k_futex futex;
        int limit;
#else
        struct k_sem kernel_sem;
#endif
};


#ifdef CONFIG_USERSPACE
#define SYS_SEM_DEFINE(_name, _initial_count, _count_limit) \
        struct sys_sem _name = { \
                .futex = { _initial_count }, \
                .limit = _count_limit \
        }; \
        BUILD_ASSERT(((_count_limit) != 0) && \
                     ((_initial_count) <= (_count_limit)))
#else
/* Stuff this in the section with the rest of the k_sem objects, since they
 * are identical and can be treated as a k_sem in the boot initialization code
 */
#define SYS_SEM_DEFINE(_name, _initial_count, _count_limit) \
        STRUCT_SECTION_ITERABLE_ALTERNATE(k_sem, sys_sem, _name) = { \
                .kernel_sem = Z_SEM_INITIALIZER(_name.kernel_sem, \
                                                _initial_count, _count_limit) \
        }; \
        BUILD_ASSERT(((_count_limit) != 0) && \
                     ((_initial_count) <= (_count_limit)))
#endif

int sys_sem_init(struct sys_sem *sem, unsigned int initial_count,
                unsigned int limit);

int sys_sem_give(struct sys_sem *sem);

int sys_sem_take(struct sys_sem *sem, k_timeout_t timeout);

unsigned int sys_sem_count_get(struct sys_sem *sem);


#if defined(__GNUC__)
static ALWAYS_INLINE void z_sys_sem_lock_onexit(__maybe_unused int *rc)
{
        __ASSERT(*rc == 1, "SYS_SEM_LOCK exited with goto, break or return, "
                           "use SYS_SEM_LOCK_BREAK instead.");
}
#define SYS_SEM_LOCK_ONEXIT __attribute__((__cleanup__(z_sys_sem_lock_onexit)))
#else
#define SYS_SEM_LOCK_ONEXIT
#endif


#define SYS_SEM_LOCK_BREAK continue

#define SYS_SEM_LOCK(sem)                                                                          \
        for (int __rc SYS_SEM_LOCK_ONEXIT = sys_sem_take((sem), K_FOREVER); ({                     \
                     __ASSERT(__rc >= 0, "Failed to take sem: %d", __rc);                          \
                     __rc == 0;                                                                    \
             });                                                                                   \
             ({                                                                                    \
                     __rc = sys_sem_give((sem));                                                   \
                     __ASSERT(__rc == 0, "Failed to give sem: %d", __rc);                          \
             }),                                                                                   \
                      __rc = 1)


#ifdef __cplusplus
}
#endif

#endif
```

## 10.2\_可见注释译文（Comments）

1. 版权归2019年的Intel Corporation所有；Apache-2.0许可证标识与原署名保持原样。

2. 启用用户模式时，sys_sem位于用户内存，为用户模式线程提供计数信号量（Counting Semaphore）；未启用用户模式时，行为类似k_sem。

3. 将此对象放入与其他k_sem对象相同的段，因为两者在此配置下布局相同，启动初始化代码可把它当作k_sem处理。

## 10.3\_源码跟踪译注

> 译注——两种布局：用户模式分支含k_futex和int limit，非用户模式分支含k_sem。静态定义宏的非用户模式版本通过STRUCT_SECTION_ITERABLE_ALTERNATE把包装对象纳入k_sem段；应用不能在两种配置之间假定结构大小和字段相同。

> 译注——计数与竞争：kernel/userspace/sem.c使用原子CAS循环。-1是竞争状态标记，不是负的资源数量或等待者人数；give可从负状态变成1并唤醒全部等待者，真正取得资源仍由take的原子操作决定。count_get将非正数报告为0。

> 译注——初始化差异：用户模式运行时初始化检查(0, INT_MAX]及初始值。两个静态宏没有同样的INT_MAX上界断言；非用户模式运行时包装忽略k_sem_init返回值。译文保留公共文档并单独指出这些实现差别，未将其描述为完全相同的错误检查。

> 译注——等待与返回：用户模式take在被唤醒或值变化后重试，循环复用timeout；非用户模式take将底层-EBUSY／-EAGAIN转为-ETIMEDOUT。用户模式give达到上限时返回-EAGAIN，非用户模式give转发后返回0。

> 译注——保护块退出：SYS_SEM_LOCK用for的递增表达式give一次；SYS_SEM_LOCK_BREAK就是continue。GNU cleanup函数只检查退出标志，不能替代释放操作。原始诊断分别意为“未正确退出保护块，请改用SYS_SEM_LOCK_BREAK”“获取信号量失败”“给出信号量失败”。

实现定位：[仓库用户模式实现](../../../kernel/userspace/sem.c)、[仓库非用户模式实现](../../../kernel/sys/sem.c)，以及[官方用户模式实现](https://github.com/zephyrproject-rtos/zephyr/blob/main/kernel/userspace/sem.c)、[官方非用户模式实现](https://github.com/zephyrproject-rtos/zephyr/blob/main/kernel/sys/sem.c)。

仓库相对定位：[include/zephyr/sys/sem.h](../../../include/zephyr/sys/sem.h)；本次核对的[官方原始头文件](https://github.com/zephyrproject-rtos/zephyr/blob/main/include/zephyr/sys/sem.h)摘要见[来源记录](../SOURCES.md)。去掉注释和展示空白后，本页代码与本次读取的原始头文件一致；不能把这种一致性推广到以后滚动更新的main。
