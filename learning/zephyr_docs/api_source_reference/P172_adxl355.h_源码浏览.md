        SENSOR_ATTR_ADXL355_ACTIVITY_ENABLE_Y,
        SENSOR_ATTR_ADXL355_ACTIVITY_ENABLE_Z,
};

enum adxl355_hpf_corner {
        ADXL355_HPF_OFF = ADXL355_DT_HPF_OFF,
