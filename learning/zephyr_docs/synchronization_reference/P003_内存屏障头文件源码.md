```c
#ifndef ZEPHYR_INCLUDE_SYS_BARRIER_H_
#define ZEPHYR_INCLUDE_SYS_BARRIER_H_

#include <zephyr/toolchain.h>

#if defined(CONFIG_BARRIER_OPERATIONS_ARCH)
# if defined(CONFIG_ARM)
# include <zephyr/arch/arm/barrier.h>
# elif defined(CONFIG_ARM64)
# include <zephyr/arch/arm64/barrier.h>
# elif defined(CONFIG_ARC)
# include <zephyr/arch/arc/barrier.h>
# endif
#elif defined(CONFIG_BARRIER_OPERATIONS_BUILTIN)
#include <zephyr/sys/barrier_builtin.h>
#endif

#ifdef __cplusplus
extern "C" {
#endif


static ALWAYS_INLINE void barrier_sync_synchronize(void)
{
#if defined(CONFIG_BARRIER_OPERATIONS_ARCH) || defined(CONFIG_BARRIER_OPERATIONS_BUILTIN)
        z_barrier_sync_synchronize();
#endif
}

static ALWAYS_INLINE void barrier_dmem_fence_full(void)
{
#if defined(CONFIG_BARRIER_OPERATIONS_ARCH) || defined(CONFIG_BARRIER_OPERATIONS_BUILTIN)
        z_barrier_dmem_fence_full();
#endif
}

static ALWAYS_INLINE void barrier_dsync_fence_full(void)
{
#if defined(CONFIG_BARRIER_OPERATIONS_ARCH) || defined(CONFIG_BARRIER_OPERATIONS_BUILTIN)
        z_barrier_dsync_fence_full();
#endif
}

static ALWAYS_INLINE void barrier_isync_fence_full(void)
{
#if defined(CONFIG_BARRIER_OPERATIONS_ARCH) || defined(CONFIG_BARRIER_OPERATIONS_BUILTIN)
        z_barrier_isync_fence_full();
#endif
}


#ifdef __cplusplus
} /* extern "C" */
#endif

#endif /* ZEPHYR_INCLUDE_SYS_ATOMIC_H_ */
```

## 3.3\_源码跟踪译注

> 译注——后端选择：CONFIG_BARRIER_OPERATIONS_ARCH分支只为Arm、Arm64和ARC包含各自头文件，内建分支包含barrier_builtin.h。四个公共包装只有在ARCH或BUILTIN已选择时调用z_barrier_*；两者均未选择时函数体为空，因此不能把每个调用都解释为一定存在硬件屏障。

> 译注——Arm实现：本次官方arch/arm/barrier.h中，sync_synchronize调用__sync_synchronize，dmem、dsync、isync分别调用__DMB、__DSB、__ISB。这是该后端的映射，不可推广为内建后端或全部架构的相同指令行为。

> 译注——机制边界：屏障控制访问与执行顺序，不会自动锁住共享对象，也不等同于缓存写回／失效或等待多个线程到齐的同步对象。文件末尾保护宏名称的注释笔误保持原样并在注释译文中说明。

Arm后端定位：[仓库源码](../../../include/zephyr/arch/arm/barrier.h)与[本次官方源码](https://github.com/zephyrproject-rtos/zephyr/blob/main/include/zephyr/arch/arm/barrier.h)。

仓库相对定位：[include/zephyr/sys/barrier.h](../../../include/zephyr/sys/barrier.h)；本次核对的[官方原始头文件](https://github.com/zephyrproject-rtos/zephyr/blob/main/include/zephyr/sys/barrier.h)摘要见[来源记录](../SOURCES.md)。去掉注释和展示空白后，本页代码与本次读取的原始头文件一致；不能把这种一致性推广到以后滚动更新的main。
