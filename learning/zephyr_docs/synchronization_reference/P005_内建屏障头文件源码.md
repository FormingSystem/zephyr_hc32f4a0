```c
#ifndef ZEPHYR_INCLUDE_SYS_BARRIER_BUILTIN_H_
#define ZEPHYR_INCLUDE_SYS_BARRIER_BUILTIN_H_

#ifndef ZEPHYR_INCLUDE_SYS_BARRIER_H_
#error Please include <zephyr/sys/barrier.h>
#endif

#include <zephyr/toolchain.h>

#ifdef __cplusplus
extern "C" {
#endif

static ALWAYS_INLINE void z_barrier_sync_synchronize(void)
{
        __sync_synchronize();
}

static ALWAYS_INLINE void z_barrier_dmem_fence_full(void)
{
#if defined(__GNUC__)
        /* GCC-ism */
        __atomic_thread_fence(__ATOMIC_SEQ_CST);
#else
        atomic_thread_fence(memory_order_seq_cst);
#endif
}

static ALWAYS_INLINE void z_barrier_dsync_fence_full(void)
{
#if defined(__GNUC__)
        /* GCC-ism */
        __atomic_thread_fence(__ATOMIC_SEQ_CST);
#else
        atomic_thread_fence(memory_order_seq_cst);
#endif
}

static ALWAYS_INLINE void z_barrier_isync_fence_full(void)
{
        __asm__ volatile("" : : : "memory");
}

#ifdef __cplusplus
}
#endif

#endif /* ZEPHYR_INCLUDE_SYS_BARRIER_BUILTIN_H_ */
```

## 5.3\_源码跟踪译注

> 译注——包含入口：文件检查ZEPHYR_INCLUDE_SYS_BARRIER_H_；缺少该公共头文件保护宏时触发#error，提示包含zephyr/sys/barrier.h。应用应使用公共入口，不直接选择此内部实现。

> 译注——数据栅栏：sync_synchronize使用__sync_synchronize。dmem与dsync在__GNUC__分支均调用__atomic_thread_fence(__ATOMIC_SEQ_CST)，否则调用atomic_thread_fence(memory_order_seq_cst)。此处的dsync回退没有独立的额外指令同步步骤。

> 译注——指令同步回退：isync使用带memory破坏描述的空volatile汇编，约束编译器对内存操作的移动；这里没有实际处理器指令来清空流水线。这与Arm架构后端调用__ISB的行为需分开理解。

仓库相对定位：[include/zephyr/sys/barrier_builtin.h](../../../include/zephyr/sys/barrier_builtin.h)；本次核对的[官方原始头文件](https://github.com/zephyrproject-rtos/zephyr/blob/main/include/zephyr/sys/barrier_builtin.h)摘要见[来源记录](../SOURCES.md)。去掉注释和展示空白后，本页代码与本次读取的原始头文件一致；不能把这种一致性推广到以后滚动更新的main。
