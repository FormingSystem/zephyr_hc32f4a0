<a id="l00078"></a> * Advance a free-running index into [0, 2N) and convert it to the index type.
<a id="l00079"></a> * Takes the buffer capacity N and forms the 2N wrap limit internally, so the
<a id="l00080"></a> * [0, 2N) convention stays in one place. Callers guarantee value &lt; 4N (index
<a id="l00081"></a> * &lt; 2N plus an increment &lt;= N), so a single conditional subtraction suffices.

<a id="l00083"></a>static ALWAYS_INLINE ring_buf_idx_t rb_idx_advance(size_t value, ring_buf_size_t size)

<a id="l00086"></a>

<a id="l00088"></a>}

<a id="l00291"></a>static inline uint32_t ring_buf_put_ptr(struct ring_buf *rb, uint8_t **data, size_t offset)
<a id="l00292"></a>{
<a id="l00293"></a>        ring_buf_idx_t write_idx = rb_load_relaxed(&amp;rb-&gt;write_idx);
<a id="l00294"></a>        ring_buf_idx_t read_idx = rb_load_acquire(&amp;rb-&gt;read_idx);
<a id="l00295"></a>        ring_buf_idx_t off, avail;
<a id="l00296"></a>
<a id="l00297"></a>        __ASSERT_NO_MSG(offset &lt;= ring_buf_space_get(rb));
<a id="l00298"></a>        if (offset &gt; 0) {
<a id="l00299"></a>                write_idx = rb_idx_advance((size_t)write_idx + offset, rb-&gt;size);
<a id="l00300"></a>        }
<a id="l00301"></a>
<a id="l00302"></a>        if (write_idx &gt;= rb-&gt;size) {
<a id="l00303"></a>                off = write_idx - rb-&gt;size;
<a id="l00304"></a>                avail = (read_idx &gt;= rb-&gt;size ? rb-&gt;size : read_idx) - off;
<a id="l00305"></a>        } else {
<a id="l00306"></a>                off = write_idx;
<a id="l00307"></a>                avail = (read_idx &gt;= rb-&gt;size ? read_idx - rb-&gt;size : rb-&gt;size) - off;
<a id="l00308"></a>        }
<a id="l00309"></a>        *data = &amp;rb-&gt;buffer[off];
<a id="l00310"></a>        return avail;
<a id="l00311"></a>}
<a id="l00312"></a>
<a id="l00322"></a>static inline void ring_buf_commit(struct ring_buf *rb, size_t size)
<a id="l00323"></a>{
<a id="l00324"></a>        size_t write_idx = rb_load_relaxed(&amp;rb-&gt;write_idx);
<a id="l00325"></a>
<a id="l00326"></a>        __ASSERT_NO_MSG(size &lt;= ring_buf_space_get(rb));
<a id="l00327"></a>        rb_store_release(&amp;rb-&gt;write_idx, rb_idx_advance(write_idx + size, rb-&gt;size));
<a id="l00328"></a>}

<a id="l00347"></a>static inline uint32_t ring_buf_get_ptr(struct ring_buf *rb, uint8_t **data, size_t offset)
<a id="l00348"></a>{
<a id="l00349"></a>        ring_buf_idx_t read_idx = rb_load_relaxed(&amp;rb-&gt;read_idx);
<a id="l00350"></a>        ring_buf_idx_t write_idx = rb_load_acquire(&amp;rb-&gt;write_idx);
<a id="l00351"></a>        ring_buf_idx_t off, avail;
<a id="l00352"></a>
<a id="l00353"></a>        __ASSERT_NO_MSG(offset &lt;= ring_buf_size_get(rb));
<a id="l00354"></a>        if (offset &gt; 0) {
<a id="l00355"></a>                read_idx = rb_idx_advance((size_t)read_idx + offset, rb-&gt;size);
<a id="l00356"></a>        }
<a id="l00357"></a>
<a id="l00358"></a>        if (read_idx &gt;= rb-&gt;size) {
<a id="l00359"></a>                off = read_idx - rb-&gt;size;
<a id="l00360"></a>                avail = (write_idx &gt;= rb-&gt;size ? write_idx : 2U * rb-&gt;size) - read_idx;
<a id="l00361"></a>        } else {
<a id="l00362"></a>                off = read_idx;
<a id="l00363"></a>                avail = (write_idx &gt;= rb-&gt;size ? rb-&gt;size : write_idx) - read_idx;
<a id="l00364"></a>        }
<a id="l00365"></a>        *data = &amp;rb-&gt;buffer[off];
<a id="l00366"></a>        return avail;
<a id="l00367"></a>}
<a id="l00368"></a>
<a id="l00378"></a>static inline void ring_buf_consume(struct ring_buf *rb, size_t size)
<a id="l00379"></a>{
<a id="l00380"></a>        size_t read_idx = rb_load_relaxed(&amp;rb-&gt;read_idx);
<a id="l00381"></a>
<a id="l00382"></a>        __ASSERT_NO_MSG(size &lt;= ring_buf_size_get(rb));
<a id="l00383"></a>        rb_store_release(&amp;rb-&gt;read_idx, rb_idx_advance(read_idx + size, rb-&gt;size));
<a id="l00384"></a>}
<a id="l00385"></a>
<a id="l00395"></a>static inline uint32_t ring_buf_put(struct ring_buf *rb, const uint8_t *data, uint32_t size)
<a id="l00396"></a>{
<a id="l00397"></a>        uint8_t *dst;
<a id="l00398"></a>        uint32_t total = 0;
<a id="l00399"></a>        uint32_t chunk;
<a id="l00400"></a>        uint32_t avail;
<a id="l00401"></a>
<a id="l00402"></a>        do {
<a id="l00403"></a>                avail = ring_buf_put_ptr(rb, &amp;dst, total);
<a id="l00404"></a>                chunk = MIN(avail, size - total);
<a id="l00405"></a>                if (chunk == 0U) {
<a id="l00406"></a>                        break;
<a id="l00407"></a>                }
<a id="l00408"></a>                memcpy(dst, &amp;data[total], chunk);
<a id="l00409"></a>                total += chunk;
<a id="l00410"></a>        } while (total &lt; size);
<a id="l00411"></a>        ring_buf_commit(rb, total);
<a id="l00412"></a>
<a id="l00413"></a>        return total;
<a id="l00414"></a>}
<a id="l00415"></a>
<a id="l00425"></a>static inline uint32_t ring_buf_get(struct ring_buf *rb, uint8_t *data, uint32_t size)
<a id="l00426"></a>{
<a id="l00427"></a>        uint8_t *src;
<a id="l00428"></a>        uint32_t chunk;
<a id="l00429"></a>        uint32_t avail;
<a id="l00430"></a>        uint32_t total = 0;
<a id="l00431"></a>
<a id="l00432"></a>        do {
<a id="l00433"></a>                avail = ring_buf_get_ptr(rb, &amp;src, total);
<a id="l00434"></a>                chunk = MIN(avail, size - total);
<a id="l00435"></a>                if (chunk == 0U) {
<a id="l00436"></a>                        break;
<a id="l00437"></a>                }
<a id="l00438"></a>                if (!IS_ENABLED(CONFIG_RING_BUFFER) || data != NULL) {
<a id="l00439"></a>                        memcpy(&amp;data[total], src, chunk);
<a id="l00440"></a>                }
<a id="l00441"></a>                total += chunk;
<a id="l00442"></a>        } while (total &lt; size);
<a id="l00443"></a>        ring_buf_consume(rb, total);
<a id="l00444"></a>
<a id="l00445"></a>        return total;
<a id="l00446"></a>}
<a id="l00447"></a>
<a id="l00460"></a>static inline uint32_t ring_buf_peek(const struct ring_buf *rb, uint8_t *data, uint32_t size)
<a id="l00461"></a>{
<a id="l00462"></a>        uint8_t *src;

## 4.4\_源码跟踪译注

> 译注——版本与编译入口：本次官方main采用仅需头文件（Header-only）的实现，lib/utils/CMakeLists.txt不再编译ring_buffer.c；CONFIG_RING_BUFFER恢复旧claim/finish与item接口，并选择DEPRECATED。仓库当前实现仍有独立ring_buffer.c，不能把本页新版布局、配置含义和性能直接套用到本地版本。

> 译注——单生产者／单消费者（Single Producer Single Consumer，SPSC）：当前官方头文件在CONFIG_SMP且CONFIG_MP_MAX_NUM_CPUS大于1时使用获取／释放（Acquire/Release）原子访问发布索引，其他配置使用相应编译器屏障（Compiler Barrier）。这不提供多生产者或多消费者的互斥，也不自动解决DMA缓存一致性；等待与唤醒仍由应用安排。指南中要求同步原语提供内存屏障的文字与此处实现细节分别保留，目标版本须以实际源码为准。

> 译注——容量边界：当前默认RING_BUFFER_MAX_SIZE为UINT16_MAX/2U，即32767字节；CONFIG_RING_BUFFER_LARGE启用后为UINT32_MAX/4U，即1073741823字节，索引与容量字段由16位增为32位。Kconfig的32KB／1GB是概括表述，不能当作可原样传入的精确上限。容量不要求为2的幂，用户可见容量无需再减一个槽位。

> 译注——零复制（Zero-copy）指针接口不改变索引，也不保存一次独占预留。offset相对尚未提交／消费的当前索引，由调用者跟踪；在同一侧没有推进索引时重复以offset=0获取指针，会指向同一逻辑区域。commit／consume发布实际处理的字节数，不是分配或释放后备内存。

> 译注——配置与文档范围：本次公开API页只列新版接口；源码浏览页还保留CONFIG_RING_BUFFER条件下的兼容实现及关闭时的编译错误宏。原网页隐藏的Doxygen注释不作为可见源码注释计数，不把这四页登记解释为所有其他历史版本API已完成翻译。

> 译注——结构布局（Structure Layout）：buffer指向后备字节存储，size为容量N；read_idx与write_idx在[0, 2N)范围内推进，用轮次区分满与空。公开结构网页隐藏了这些内部字段，本页按源码保留；不要自行修改索引。put_claimed和get_claimed仅在兼容配置下存在。

> 译注——默认get与兼容get对NULL目标的处理不同；默认路径仍会复制。初始化／复位只清索引，不清内存。编译错误宏中的迁移提示完整保留：旧接口在CONFIG_RING_BUFFER=n时不可用，过渡期可以启用该配置，或按照Zephyr 4.5迁移指南切换替代API。
