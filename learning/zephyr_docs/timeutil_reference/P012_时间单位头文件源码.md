<a id="l00046"></a>
<a id="l00047"></a>/* Exhaustively enumerated, highly optimized time unit conversion API */
<a id="l00048"></a>
<a id="l00049"></a>#if defined(CONFIG_TIMER_READS_ITS_FREQUENCY_AT_RUNTIME) || \

<a id="l00069"></a>        #if defined(CONFIG_TIMER_READS_ITS_FREQUENCY_AT_RUNTIME) || \

<a id="l00101"></a>#define z_tmcvt_use_fast_algo(from_hz, to_hz) \
<a id="l00102"></a>        ((DIV_ROUND_UP(CONFIG_SYS_CLOCK_MAX_TIMEOUT_DAYS * 24ULL * 3600ULL * from_hz, \
<a id="l00103"></a>                           UINT32_MAX) * to_hz) &lt;= UINT32_MAX)
<a id="l00104"></a>
<a id="l00105"></a>/* Time converter generator gadget.  Selects from one of three
<a id="l00106"></a> * conversion algorithms: ones that take advantage when the
<a id="l00107"></a> * frequencies are an integer ratio (in either direction), or a full
<a id="l00108"></a> * precision conversion.  Clever use of extra arguments causes all the
<a id="l00109"></a> * selection logic to be optimized out, and the generated code even
<a id="l00110"></a> * reduces to 32 bit only if a ratio conversion is available and the
<a id="l00111"></a> * result is 32 bits.
<a id="l00112"></a> *
<a id="l00113"></a> * This isn't intended to be used directly, instead being wrapped
<a id="l00114"></a> * appropriately in a user-facing API.  The boolean arguments are:
<a id="l00115"></a> *
<a id="l00116"></a> *    const_hz  - The hz arguments are known to be compile-time
<a id="l00117"></a> *                constants (because otherwise the modulus test would
<a id="l00118"></a> *                have to be done at runtime)
<a id="l00119"></a> *    result32  - The result will be truncated to 32 bits on use
<a id="l00120"></a> *    round_up  - Return the ceiling of the resulting fraction
<a id="l00121"></a> *    round_off - Return the nearest value to the resulting fraction
<a id="l00122"></a> *                (pass both round_up/off as false to get "round_down")
<a id="l00123"></a> *
<a id="l00124"></a> * All of this must be implemented as expressions so that, when constant,
<a id="l00125"></a> * the results may be used to initialize global variables.
<a id="l00126"></a> */
<a id="l00127"></a>
<a id="l00128"></a>/* true if the conversion is the identity */
<a id="l00129"></a>#define z_tmcvt_is_identity(__from_hz, __to_hz) \
<a id="l00130"></a>        ((__to_hz) == (__from_hz))
<a id="l00131"></a>
<a id="l00132"></a>/* true if the conversion requires a simple integer multiply */
<a id="l00133"></a>#define z_tmcvt_is_int_mul(__from_hz, __to_hz) \
<a id="l00134"></a>        ((__to_hz) &gt; (__from_hz) &amp;&amp; (__to_hz) % (__from_hz) == 0U)
<a id="l00135"></a>
<a id="l00136"></a>/* true if the conversion requires a simple integer division */
<a id="l00137"></a>#define z_tmcvt_is_int_div(__from_hz, __to_hz) \
<a id="l00138"></a>        ((__from_hz) &gt; (__to_hz) &amp;&amp; (__from_hz) % (__to_hz) == 0U)
<a id="l00139"></a>
<a id="l00140"></a>/*
<a id="l00141"></a> * Compute the offset needed to round the result correctly when
<a id="l00142"></a> * the conversion requires a simple integer division
<a id="l00143"></a> */
<a id="l00144"></a>#define z_tmcvt_off_div(__from_hz, __to_hz, __round_up, __round_off)    \
<a id="l00145"></a>        ((__round_off) ? ((__from_hz) / (__to_hz)) / 2 :                \
<a id="l00146"></a>         (__round_up) ? ((__from_hz) / (__to_hz)) - 1 :                 \
<a id="l00147"></a>         0)
<a id="l00148"></a>
<a id="l00149"></a>/*
<a id="l00150"></a> * All users of this macro MUST ensure its output is never used when a/b
<a id="l00151"></a> * is zero because it incorrectly but by design never returns zero.
<a id="l00152"></a> *
<a id="l00153"></a> * Some compiler versions emit a divide-by-zero warning for this code:
<a id="l00154"></a> * "false ? 42/0 : 43". Dealing with (generated) dead code is hard:
<a id="l00155"></a> * https://github.com/zephyrproject-rtos/zephyr/issues/63564
<a id="l00156"></a> * https://blog.llvm.org/2011/05/what-every-c-programmer-should-know_21.html
<a id="l00157"></a> *
<a id="l00158"></a> * To silence such divide-by-zero warnings, "cheat" and never return
<a id="l00159"></a> * zero.  Return 1 instead. Use octal "01u" as a breadcrumb to ease a
<a id="l00160"></a> * little bit the huge pain of "reverse-engineering" pre-processor
<a id="l00161"></a> * output.
<a id="l00162"></a> *
<a id="l00163"></a> * The "Elvis" operator "a/b ?: 1" is tempting because it avoids
<a id="l00164"></a> * evaluating the same expression twice. However: 1. it's a non-standard
<a id="l00165"></a> * GNU extension; 2. everything in this file is designed to be computed
<a id="l00166"></a> * at compile time anyway.
<a id="l00167"></a> */
<a id="l00168"></a>#define z_tmcvt_divisor(a, b) ((a)/(b) ? (a)/(b) : 1U)
<a id="l00169"></a>
<a id="l00170"></a>/*
<a id="l00171"></a> * Compute the offset needed to round the result correctly when
<a id="l00172"></a> * the conversion requires a full mul/div
<a id="l00173"></a> */
<a id="l00174"></a>#define z_tmcvt_off_gen(__from_hz, __to_hz, __round_up, __round_off)    \
<a id="l00175"></a>        ((__round_off) ? (__from_hz) / 2 :                              \
<a id="l00176"></a>         (__round_up) ? (__from_hz) - 1 :                               \
<a id="l00177"></a>         0)
<a id="l00178"></a>
<a id="l00179"></a>/* Integer division 32-bit conversion */
<a id="l00180"></a>#define z_tmcvt_int_div_32(__t, __from_hz, __to_hz, __round_up, __round_off) \
<a id="l00181"></a>        ((uint64_t) (__t) &lt;= 0xffffffffU -                              \
<a id="l00182"></a>         z_tmcvt_off_div(__from_hz, __to_hz, __round_up, __round_off) ? \
<a id="l00183"></a>         ((uint32_t)(((__t) +                                           \
<a id="l00184"></a>                     z_tmcvt_off_div(__from_hz, __to_hz,                \
<a id="l00185"></a>                                     __round_up, __round_off)) /        \
<a id="l00186"></a>          z_tmcvt_divisor(__from_hz, __to_hz)))                         \
<a id="l00187"></a>         :                                                              \
<a id="l00188"></a>         (uint32_t) (((uint64_t) (__t) +                                \
<a id="l00189"></a>                      z_tmcvt_off_div(__from_hz, __to_hz,               \
<a id="l00190"></a>                                      __round_up, __round_off)) /       \
<a id="l00191"></a>                     z_tmcvt_divisor(__from_hz, __to_hz))               \
<a id="l00192"></a>                )
<a id="l00193"></a>
<a id="l00194"></a>/* Integer multiplication 32-bit conversion */
<a id="l00195"></a>#define z_tmcvt_int_mul_32(__t, __from_hz, __to_hz)     \
<a id="l00196"></a>        ((uint32_t) ((__t)*((__to_hz) / (__from_hz))))
<a id="l00197"></a>
<a id="l00198"></a>/* General 32-bit conversion */
<a id="l00199"></a>#define z_tmcvt_gen_32(__t, __from_hz, __to_hz, __round_up, __round_off) \
<a id="l00200"></a>        ((uint32_t) (((uint64_t) (__t)*(__to_hz) +                      \
<a id="l00201"></a>                      z_tmcvt_off_gen(__from_hz, __to_hz, __round_up, __round_off)) / (__from_hz)))
<a id="l00202"></a>
<a id="l00203"></a>/* Integer division 64-bit conversion */

<a id="l00205"></a>        (((uint64_t) (__t) + z_tmcvt_off_div(__from_hz, __to_hz,        \
<a id="l00206"></a>                                             __round_up, __round_off)) / \
<a id="l00207"></a>        z_tmcvt_divisor(__from_hz, __to_hz))
<a id="l00208"></a>
<a id="l00209"></a>/* Integer multiplication 64-bit conversion */
<a id="l00210"></a>#define z_tmcvt_int_mul_64(__t, __from_hz, __to_hz)     \
<a id="l00211"></a>        (uint64_t) (__t)*((__to_hz) / (__from_hz))
<a id="l00212"></a>
<a id="l00213"></a>/* Fast 64-bit conversion. This relies on the multiply not overflowing */
<a id="l00214"></a>#define z_tmcvt_gen_64_fast(__t, __from_hz, __to_hz, __round_up, __round_off) \
<a id="l00215"></a>        (((uint64_t) (__t)*(__to_hz) + \
<a id="l00216"></a>          z_tmcvt_off_gen(__from_hz, __to_hz, __round_up, __round_off)) / (__from_hz))
<a id="l00217"></a>
<a id="l00218"></a>/* Slow 64-bit conversion. This avoids overflowing the multiply */
<a id="l00219"></a>#define z_tmcvt_gen_64_slow(__t, __from_hz, __to_hz, __round_up, __round_off) \
<a id="l00220"></a>        ((((uint64_t) (__t) / (__from_hz))*(__to_hz)) +                 \
<a id="l00221"></a>         (((((uint64_t) (__t) % (__from_hz))*(__to_hz)) +               \
<a id="l00222"></a>          z_tmcvt_off_gen(__from_hz, __to_hz, __round_up, __round_off)) / (__from_hz)))
<a id="l00223"></a>
<a id="l00224"></a>/* General 64-bit conversion. Uses one of the two above macros */
<a id="l00225"></a>#define z_tmcvt_gen_64(__t, __from_hz, __to_hz, __round_up, __round_off) \
<a id="l00226"></a>        (z_tmcvt_use_fast_algo(__from_hz, __to_hz) ?                    \
<a id="l00227"></a>         z_tmcvt_gen_64_fast(__t, __from_hz, __to_hz, __round_up, __round_off) : \
<a id="l00228"></a>         z_tmcvt_gen_64_slow(__t, __from_hz, __to_hz, __round_up, __round_off))
<a id="l00229"></a>
<a id="l00230"></a>/* Convert, generating a 32-bit result */
<a id="l00231"></a>#define z_tmcvt_32(__t, __from_hz, __to_hz, __const_hz, __round_up, __round_off) \
<a id="l00232"></a>        ((__const_hz) ?                                                 \
<a id="l00233"></a>         (                                                              \
<a id="l00234"></a>                 z_tmcvt_is_identity(__from_hz, __to_hz) ?              \
<a id="l00235"></a>                 (uint32_t) (__t)                                       \
<a id="l00236"></a>                 :                                                      \
<a id="l00237"></a>                 z_tmcvt_is_int_div(__from_hz, __to_hz) ?               \
<a id="l00238"></a>                 z_tmcvt_int_div_32(__t, __from_hz, __to_hz, __round_up, __round_off) \
<a id="l00239"></a>                 :                                                      \
<a id="l00240"></a>                 z_tmcvt_is_int_mul(__from_hz, __to_hz) ?               \
<a id="l00241"></a>                 z_tmcvt_int_mul_32(__t, __from_hz, __to_hz)            \
<a id="l00242"></a>                 :                                                      \
<a id="l00243"></a>                 z_tmcvt_gen_32(__t, __from_hz, __to_hz, __round_up, __round_off) \
<a id="l00244"></a>                 )                                                      \
<a id="l00245"></a>         :                                                              \
<a id="l00246"></a>         z_tmcvt_gen_32(__t, __from_hz, __to_hz, __round_up, __round_off) \
<a id="l00247"></a>                )
<a id="l00248"></a>
<a id="l00249"></a>/* Convert, generating a 64-bit result */
<a id="l00250"></a>#define z_tmcvt_64(__t, __from_hz, __to_hz, __const_hz, __round_up, __round_off) \
<a id="l00251"></a>        ((__const_hz) ?                                                 \
<a id="l00252"></a>         (                                                              \
<a id="l00253"></a>                 z_tmcvt_is_identity(__from_hz, __to_hz) ?              \
<a id="l00254"></a>                 (uint64_t) (__t)                                       \
<a id="l00255"></a>                 :                                                      \
<a id="l00256"></a>                 z_tmcvt_is_int_div(__from_hz, __to_hz) ?               \
<a id="l00257"></a>                 z_tmcvt_int_div_64(__t, __from_hz, __to_hz, __round_up, __round_off) \
<a id="l00258"></a>                 :                                                      \
<a id="l00259"></a>                 z_tmcvt_is_int_mul(__from_hz, __to_hz) ?               \
<a id="l00260"></a>                 z_tmcvt_int_mul_64(__t, __from_hz, __to_hz)            \
<a id="l00261"></a>                 :                                                      \
<a id="l00262"></a>                 z_tmcvt_gen_64(__t, __from_hz, __to_hz, __round_up, __round_off) \
<a id="l00263"></a>                 )                                                      \
<a id="l00264"></a>         :                                                              \
<a id="l00265"></a>         z_tmcvt_gen_64_slow(__t, __from_hz, __to_hz, __round_up, __round_off) \
<a id="l00266"></a>                )
<a id="l00267"></a>
<a id="l00268"></a>#define z_tmcvt(__t, __from_hz, __to_hz, __const_hz, __result32, __round_up, __round_off) \

> 译注——接口名中的sec／ms／us／ns分别为秒、毫秒、微秒和纳秒，ticks为内核时钟节拍（Kernel Ticks），cyc为硬件计时周期（Hardware Cycles）。硬件计时频率不一定等于CPU内核频率；运行时更新频率也不会自动修正早先采集的周期计数。

> 译注——floor截去非负结果的小数部分，ceil向上取整，near取最近整数；当前偏移算法对恰好半单位的非负值向上舍入。32／64后缀表示返回宽度，32位结果可能截断，64位也不保证任意输入不溢出。这些宏用于无符号时长，不直接接受负数或编码后的绝对超时值。

> 译注——已知频率比时可选择恒等、整数乘法或整数除法，否则执行一般乘除换算。CONFIG_SYS_CLOCK_MAX_TIMEOUT_DAYS参与快速算法选择，是实现的范围假设而非通用溢出检测；慢速算法拆分乘法以避免该中间乘积溢出，输出仍需落在可表达范围内。

> 译注——SYS_FOREVER_MS／SYS_FOREVER_US的-1是对应单位的无限等待哨兵，不是可任意参与无符号换算的普通时长，也不是完整的k_timeout_t。毫秒到内核超时应按SYS_TIMEOUT_MS等接口约定处理；不能把其他负数一律解释为永久等待。

> 译注——源码中的z_tmcvt_divisor故意不返回0，用于避免生成的未执行分支触发编译器除零警告，调用者不能在a/b为0时使用其结果。原注释提到八进制01u，当前代码实际采用1U；原注释与代码均保持原样。

> 译注——生成器历史注释称纳秒／微秒转换仅有64位版本，但当前头文件确实包含32位变体，例如k_ns_to_ticks_ceil32。原注释保留并在此注明差异；可用接口以当前声明为准。原始Perl生成器和全部116个宏展开块完整保留，不把重复接口合并成概述。
