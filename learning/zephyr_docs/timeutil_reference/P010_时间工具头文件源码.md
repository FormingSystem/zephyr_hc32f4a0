<a id="l00383"></a>        __ASSERT_NO_MSG(ts != NULL);
<a id="l00384"></a>

<a id="l00441"></a>static inline bool timespec_add(struct timespec *a, const struct timespec *b)
<a id="l00442"></a>{
<a id="l00443"></a>        __ASSERT_NO_MSG((a != NULL) &amp;&amp; timespec_is_valid(a));
<a id="l00444"></a>        __ASSERT_NO_MSG((b != NULL) &amp;&amp; timespec_is_valid(b));
<a id="l00445"></a>
<a id="l00446"></a>#if defined(CONFIG_SPEED_OPTIMIZATIONS) &amp;&amp; HAS_BUILTIN(__builtin_add_overflow)
<a id="l00447"></a>
<a id="l00448"></a>        return !__builtin_add_overflow(a-&gt;tv_sec, b-&gt;tv_sec, &amp;a-&gt;tv_sec) &amp;&amp;
<a id="l00449"></a>               !__builtin_add_overflow(a-&gt;tv_nsec, b-&gt;tv_nsec, &amp;a-&gt;tv_nsec) &amp;&amp;
<a id="l00450"></a>               timespec_normalize(a);
<a id="l00451"></a>
<a id="l00452"></a>#else
<a id="l00453"></a>
<a id="l00454"></a>        if ((a-&gt;tv_sec &lt; 0) &amp;&amp; (b-&gt;tv_sec &lt; 0) &amp;&amp; (SYS_TIME_T_MIN - a-&gt;tv_sec &gt; b-&gt;tv_sec)) {
<a id="l00455"></a>                /* negative integer overflow would occur */
<a id="l00456"></a>                return false;
<a id="l00457"></a>        }
<a id="l00458"></a>
<a id="l00459"></a>        if ((a-&gt;tv_sec &gt; 0) &amp;&amp; (b-&gt;tv_sec &gt; 0) &amp;&amp; (SYS_TIME_T_MAX - a-&gt;tv_sec &lt; b-&gt;tv_sec)) {
<a id="l00460"></a>                /* positive integer overflow would occur */
<a id="l00461"></a>                return false;
<a id="l00462"></a>        }
<a id="l00463"></a>
<a id="l00464"></a>        a-&gt;tv_sec += b-&gt;tv_sec;
<a id="l00465"></a>        a-&gt;tv_nsec += b-&gt;tv_nsec;
<a id="l00466"></a>
<a id="l00467"></a>        return timespec_normalize(a);
<a id="l00468"></a>
<a id="l00469"></a>#endif
<a id="l00470"></a>}
<a id="l00471"></a>
<a id="l00484"></a>static inline bool timespec_negate(struct timespec *ts)
<a id="l00485"></a>{
<a id="l00486"></a>        __ASSERT_NO_MSG((ts != NULL) &amp;&amp; timespec_is_valid(ts));
<a id="l00487"></a>
<a id="l00488"></a>#if defined(CONFIG_SPEED_OPTIMIZATIONS) &amp;&amp; HAS_BUILTIN(__builtin_sub_overflow)
<a id="l00489"></a>
<a id="l00490"></a>        return !__builtin_sub_overflow(0LL, ts-&gt;tv_sec, &amp;ts-&gt;tv_sec) &amp;&amp;
<a id="l00491"></a>               !__builtin_sub_overflow(0L, ts-&gt;tv_nsec, &amp;ts-&gt;tv_nsec) &amp;&amp; timespec_normalize(ts);
<a id="l00492"></a>
<a id="l00493"></a>#else
<a id="l00494"></a>
<a id="l00495"></a>        if (ts-&gt;tv_sec == SYS_TIME_T_MIN) {
<a id="l00496"></a>                /* -SYS_TIME_T_MIN &gt; SYS_TIME_T_MAX, so positive integer overflow would occur */
<a id="l00497"></a>                return false;
<a id="l00498"></a>        }
<a id="l00499"></a>
<a id="l00500"></a>        ts-&gt;tv_sec = -ts-&gt;tv_sec;
<a id="l00501"></a>        ts-&gt;tv_nsec = -ts-&gt;tv_nsec;
<a id="l00502"></a>
<a id="l00503"></a>        return timespec_normalize(ts);
<a id="l00504"></a>
<a id="l00505"></a>#endif
<a id="l00506"></a>}

<a id="l00523"></a>static inline bool timespec_sub(struct timespec *a, const struct timespec *b)
<a id="l00524"></a>{
<a id="l00525"></a>        __ASSERT_NO_MSG(a != NULL);
<a id="l00526"></a>        __ASSERT_NO_MSG(b != NULL);
<a id="l00527"></a>
<a id="l00528"></a>        struct timespec neg = *b;
<a id="l00529"></a>
<a id="l00530"></a>        return timespec_negate(&amp;neg) &amp;&amp; timespec_add(a, &amp;neg);
<a id="l00531"></a>}
<a id="l00532"></a>
<a id="l00545"></a>static inline int timespec_compare(const struct timespec *a, const struct timespec *b)
<a id="l00546"></a>{
<a id="l00547"></a>        __ASSERT_NO_MSG((a != NULL) &amp;&amp; timespec_is_valid(a));
<a id="l00548"></a>        __ASSERT_NO_MSG((b != NULL) &amp;&amp; timespec_is_valid(b));
<a id="l00549"></a>
<a id="l00550"></a>        return (((a-&gt;tv_sec == b-&gt;tv_sec) &amp;&amp; (a-&gt;tv_nsec &lt; b-&gt;tv_nsec)) * -1) +
<a id="l00551"></a>               (((a-&gt;tv_sec == b-&gt;tv_sec) &amp;&amp; (a-&gt;tv_nsec &gt; b-&gt;tv_nsec)) * 1) +
<a id="l00552"></a>               ((a-&gt;tv_sec &lt; b-&gt;tv_sec) * -1) + ((a-&gt;tv_sec &gt; b-&gt;tv_sec));
<a id="l00553"></a>}

<a id="l00596"></a>static inline void timespec_from_timeout(k_timeout_t timeout, struct timespec *ts)
<a id="l00597"></a>{
<a id="l00598"></a>        __ASSERT_NO_MSG(ts != NULL);
<a id="l00599"></a>        __ASSERT_NO_MSG(Z_IS_TIMEOUT_RELATIVE(timeout) ||
<a id="l00600"></a>                        (IS_ENABLED(CONFIG_TIMEOUT_64BIT) &amp;&amp;
<a id="l00601"></a>                         K_TIMEOUT_EQ(timeout, (k_timeout_t){K_TICKS_FOREVER})));
<a id="l00602"></a>
<a id="l00603"></a>        /* equivalent of K_FOREVER without including kernel.h */
<a id="l00604"></a>        if (K_TIMEOUT_EQ(timeout, (k_timeout_t){K_TICKS_FOREVER})) {
<a id="l00605"></a>                /* duration == K_TICKS_FOREVER ticks */
<a id="l00606"></a>                *ts = SYS_TIMESPEC_FOREVER;
<a id="l00607"></a>                /* equivalent of K_NO_WAIT without including kernel.h */
<a id="l00608"></a>        } else if (K_TIMEOUT_EQ(timeout, (k_timeout_t){0})) {
<a id="l00609"></a>                /* duration &lt;= 0 ticks */
<a id="l00610"></a>                *ts = SYS_TIMESPEC_NO_WAIT;
<a id="l00611"></a>        } else {
<a id="l00612"></a>                *ts = SYS_TICKS_TO_TIMESPEC(timeout.ticks);
<a id="l00613"></a>        }
<a id="l00614"></a>
<a id="l00615"></a>        __ASSERT_NO_MSG(timespec_is_valid(ts));
<a id="l00616"></a>}
<a id="l00617"></a>
<a id="l00647"></a>static inline k_timeout_t timespec_to_timeout(const struct timespec *req, struct timespec *rem)
<a id="l00648"></a>{
<a id="l00649"></a>        k_timeout_t timeout;
<a id="l00650"></a>        struct timespec temp = SYS_TIMESPEC_NO_WAIT;
<a id="l00651"></a>
<a id="l00652"></a>        __ASSERT_NO_MSG((req != NULL) &amp;&amp; timespec_is_valid(req));
<a id="l00653"></a>
<a id="l00654"></a>        if (timespec_compare(req, &amp;temp) &lt;= 0) {
<a id="l00655"></a>                if (rem != NULL) {
<a id="l00656"></a>                        *rem = *req;
<a id="l00657"></a>                }
<a id="l00658"></a>                /* equivalent of K_NO_WAIT without including kernel.h */
<a id="l00659"></a>                timeout.ticks = 0;
<a id="l00660"></a>                return timeout;
<a id="l00661"></a>        }
<a id="l00662"></a>
<a id="l00663"></a>        temp = SYS_TIMESPEC_FOREVER;
<a id="l00664"></a>
<a id="l00665"></a>        if (timespec_compare(req, &amp;temp) == 0) {
<a id="l00666"></a>                if (rem != NULL) {
<a id="l00667"></a>                        *rem = SYS_TIMESPEC_NO_WAIT;
<a id="l00668"></a>                }
<a id="l00669"></a>                /* equivalent of K_FOREVER without including kernel.h */
<a id="l00670"></a>                timeout.ticks = K_TICKS_FOREVER;
<a id="l00671"></a>                return timeout;
<a id="l00672"></a>        }
<a id="l00673"></a>
<a id="l00674"></a>        temp = SYS_TIMESPEC_MAX;
<a id="l00675"></a>
<a id="l00676"></a>        if (timespec_compare(req, &amp;temp) &gt;= 0) {
<a id="l00677"></a>                /* round down to align to max ticks */
<a id="l00678"></a>                timeout.ticks = K_TICK_MAX;
<a id="l00679"></a>        } else {
<a id="l00680"></a>                /* round up to align to next tick boundary */
<a id="l00681"></a>                timeout.ticks = CLAMP(k_ns_to_ticks_ceil64(req-&gt;tv_nsec) +
<a id="l00682"></a>                                              k_sec_to_ticks_ceil64(req-&gt;tv_sec),
<a id="l00683"></a>                                      K_TICK_MIN, K_TICK_MAX);
<a id="l00684"></a>        }
<a id="l00685"></a>
<a id="l00686"></a>        if (rem != NULL) {
<a id="l00687"></a>                timespec_from_timeout(timeout, rem);
<a id="l00688"></a>                timespec_sub(rem, req);
<a id="l00689"></a>                timespec_negate(rem);
<a id="l00690"></a>        }
<a id="l00691"></a>
<a id="l00692"></a>        return timeout;
<a id="l00693"></a>}

<a id="l00700"></a>}

## 10.4\_源码跟踪译注

> 译注——阅读边界：时间表示（Time Representation）处理UTC日历字段与POSIX秒数，时间同步（Time Synchronization）映射两个计数时钟，单位换算（Unit Conversion）处理时长的单位与舍入，timespec工具处理秒／纳秒二元组。它们不自动校准RTC、选择时区或建立网络授时服务。

> 译注——版本与定位：以下译注核对本次官方main源码，链接中的本仓库源码只是相对定位入口，不保证读者的本地版本与滚动更新的latest相同。官方原文、原始代码及法律注释保留；译注不修改接口契约。

> 译注——timeutil_timegm64按tm_year+1900、tm_mon+1和日期字段计算UTC对应的POSIX秒数，不读取时区或夏令时，也不依赖tm_yday／tm_wday。实现没有空指针、日期字段合法性或任意越界字段规范化检查；64位结果也不是对无效输入的验证。

> 译注——timeutil_timegm将64位结果转换为time_t，并把errno清零后执行范围检查；当前实现仅在time_t与int32_t同宽时检查有符号32位范围，越界返回-1并置ERANGE。合法的纪元前一秒也可能返回-1，须结合errno判断。

> 译注——先将同步状态（Synchronization State）清零并设置cfg，使配置对象在使用期间有效。ref_Hz与local_Hz应为正；参考计数ref的0被用作未初始化哨兵，本地计数local允许为0。接口没有为所有配置指针和频率提供运行时有效性检查。

> 译注——state_update安装首个有效基准时总把skew设为1.0f，与原说明中“仅当skew为0时设置”的措辞存在差异。后续采样只与base比较，不与已有latest比较；只要晚于base，也可能用比原latest更早的采样替换latest。更新采样不会自动估计或应用skew。

> 译注——estimate_skew计算参考增量×本地标称频率÷本地增量÷参考标称频率，返回估计值而不修改状态。set_skew只检查skew>0；调用者还应保证有限值（Finite Value）及有效基准。它不验证新基准的ref非零或时间顺序。

> 译注——CONFIG_TIMEUTIL_APPLY_SKEW决定换算是否应用校正：本地到参考方向乘skew，反方向除skew。返回值1仅表示存储的skew不等于1，即使关闭该配置也可能返回1，不能单凭返回值认定已应用校正。

> 译注——换算使用有符号64位中间量，未覆盖所有差值、乘法和类型转换溢出。参考结果为负时返回-ERANGE且不写结果；反向结果类型为int64_t，允许负本地时间。调用者应约束计数跨度、频率与输出范围，不能按任意uint64_t全范围外推。

> 译注——skew_to_ppb按(1-skew)×10^9转换，正误差表示本地时钟偏快。超出int32_t可表达范围时返回INT32_MIN；它没有完整处理NaN／无穷值的输入验证。float分辨率也限制校正精度，不能把ppb单位理解为达到1 ppb测量精度。

> 译注——规范化（Normalization）要求0≤tv_nsec<10^9，秒字段可以为负。timespec_is_valid只检查纳秒范围；它不判断是否为正时长。normalize把溢出的纳秒进位或借位到秒；原描述使用大于NSEC_PER_SEC，当前代码实际在大于等于时进位。

> 译注——add、negate和compare要求输入已规范化；sub通过negate及add也继承这一前提。equal只比较两个原始字段，数值等价但表示未规范化的二元组未必相等。断言（Assertion）不能替代应用的输入检查。

> 译注——算术接口返回false不一律保证目标对象不变，例如内建溢出操作或相加后规范化失败可能已写入字段。调用者需要保留原值时，应对临时副本操作，成功后再采用结果；也须避免原始纳秒为最小long等导致中间取负无法表示的输入。

> 译注——SYS_TIMESPEC对秒字段做范围夹取（Clamp），纳秒字段只是转换，不能代替规范化。宏参数可能被重复求值，避免传入自增或带其他副作用的表达式。

> 译注——timespec_from_timeout接受相对超时（Relative Timeout）及永久等待哨兵；绝对期限（Absolute Deadline）不属于该接口的常规输入。普通节拍按向下换算分解为秒与纳秒；K_FOREVER映射到专用timespec值。

> 译注——timespec_to_timeout把正时长向上对齐到节拍边界，并限制到可表达的最大节拍数。rem按“请求值减换算后时长”计算，向上舍入时可能为负，封顶时可能为正；它不是一律非负的余数。rem与req必须采用不同存储，当前实现会先写rem再读取req计算差值，重叠会破坏输入。
