# 第149章\_核间数据包缓冲区完整头文件

原文：[pbuf_8h_source](https://docs.zephyrproject.org/latest/doxygen/html/pbuf_8h_source.html)。读取日期：2026-10-13，`latest`。

本页保留`zephyr/ipc/pbuf.h`的完整官方可见源码（Source Code），仅去除网页行号与行尾空白。版权为2023年 Nordic Semiconductor ASA，许可证标识（License Identifier）为`Apache-2.0`；原文保留在代码中。完整接口说明见[核间数据包缓冲区接口](P144_核间数据包缓冲区接口.md)。

阅读说明：源码中的`CONFIG_`名称是 Kconfig 构建配置系统（Build Configuration System）产生的配置宏；`CONFIG_ARCH_POSIX`对应原生模拟配置分支。`MAX`取较大值，`ROUND_UP`向上取整（Round Up），`BUILD_ASSERT`进行构建断言（Build Assertion），`IS_PTR_ALIGNED_BYTES`检查字节对齐（Byte Alignment）。CPU 为中央处理器（Central Processing Unit），TODO 为待办事项（To Do）。其余缓冲区宏、结构和函数逐项说明见前面的接口与结构章节。

## 149.1\_完整源码

```c
/*
 * Copyright (c) 2023 Nordic Semiconductor ASA
 *
 * SPDX-License-Identifier: Apache-2.0
 */

#ifndef ZEPHYR_INCLUDE_IPC_PBUF_H_
#define ZEPHYR_INCLUDE_IPC_PBUF_H_

#include <zephyr/cache.h>
#include <zephyr/devicetree.h>

#ifdef __cplusplus
extern "C" {
#endif


#define PBUF_PACKET_LEN_SZ sizeof(uint32_t)

/* Amount of data that is left unused to distinguish between empty and full. */
#define _PBUF_IDX_SIZE sizeof(uint32_t)

/* Minimal length of the data field in the buffer to store the smalest packet
 * possible.
 * (+1) for at least one byte of data.
 * (+_PBUF_IDX_SIZE) to distinguish buffer full and buffer empty.
 * Rounded up to keep wr/rd indexes pointing to aligned address.
 */
#define _PBUF_MIN_DATA_LEN ROUND_UP(PBUF_PACKET_LEN_SZ + 1 + _PBUF_IDX_SIZE, _PBUF_IDX_SIZE)

#if defined(CONFIG_ARCH_POSIX)
/* For the native simulated boards we need to modify some pointers at init */
#define PBUF_MAYBE_CONST
#else
#define PBUF_MAYBE_CONST const
#endif

struct pbuf_cfg {
        volatile uint32_t *rd_idx_loc;   /* Address of the variable holding
                                          * index value of the first valid byte
                                          * in data[].
                                          */
        volatile uint32_t *handshake_loc;/* Address of the variable holding
                                          * handshake information.
                                          */
        volatile uint32_t *wr_idx_loc;   /* Address of the variable holding
                                          * index value of the first free byte
                                          * in data[].
                                          */
        uint32_t dcache_alignment;       /* CPU data cache line size in bytes.
                                          * Used for validation - TODO: To be
                                          * replaced by flags.
                                          */
        uint32_t len;                    /* Length of data[] in bytes. */
        uint8_t *data_loc;               /* Location of the data[]. */
};

struct pbuf_data {
        volatile uint32_t wr_idx;       /* Index of the first holding first
                                         * free byte in data[]. Used for
                                         * writing.
                                         */
        volatile uint32_t rd_idx;       /* Index of the first holding first
                                         * valid byte in data[]. Used for
                                         * reading.
                                         */
};


struct pbuf {
        PBUF_MAYBE_CONST struct pbuf_cfg *const cfg; /* Configuration of the
                                                      * buffer.
                                                      */
        struct pbuf_data data;                  /* Data used to read and write
                                                 * to the buffer
                                                 */
};

#define PBUF_CFG_INIT(mem_addr, size, dcache_align, use_handshake)                      \
{                                                                                       \
        .rd_idx_loc = (uint32_t *)(mem_addr),                                           \
        .handshake_loc = use_handshake ? (uint32_t *)((uint8_t *)(mem_addr) +           \
                                _PBUF_IDX_SIZE) : NULL,                                 \
        .wr_idx_loc = (uint32_t *)((uint8_t *)(mem_addr) + MAX(dcache_align,            \
                                (use_handshake ? 2 : 1) * _PBUF_IDX_SIZE)),             \
        .data_loc = (uint8_t *)((uint8_t *)(mem_addr) +                                 \
                                MAX(dcache_align, (use_handshake ? 2 : 1) *             \
                                                  _PBUF_IDX_SIZE) + _PBUF_IDX_SIZE),    \
        .len = (uint32_t)((uint32_t)(size) - MAX(dcache_align,                          \
                        (use_handshake ? 2 : 1) * _PBUF_IDX_SIZE) - _PBUF_IDX_SIZE),    \
        .dcache_alignment = (dcache_align),                                             \
}

#define PBUF_HEADER_OVERHEAD(dcache_align) \
        (MAX(dcache_align, _PBUF_IDX_SIZE) + _PBUF_IDX_SIZE)

#define PBUF_DEFINE(name, mem_addr, size, dcache_align, use_handshake, compatibility)   \
        BUILD_ASSERT(dcache_align >= 0,                                                 \
                        "Cache line size must be non negative.");                       \
        BUILD_ASSERT((size) > 0 && IS_PTR_ALIGNED_BYTES(size, _PBUF_IDX_SIZE),          \
                        "Incorrect size.");                                             \
        BUILD_ASSERT(IS_PTR_ALIGNED_BYTES(mem_addr, MAX(dcache_align, _PBUF_IDX_SIZE)), \
                        "Misaligned memory.");                                          \
        BUILD_ASSERT(size >= (MAX(dcache_align, _PBUF_IDX_SIZE) + _PBUF_IDX_SIZE +      \
                        _PBUF_MIN_DATA_LEN), "Insufficient size.");                     \
        BUILD_ASSERT(!(compatibility) || (dcache_align) >= 8,                           \
                "Data cache alignment must be at least 8 if compatibility is enabled.");\
        static PBUF_MAYBE_CONST struct pbuf_cfg cfg_##name =                            \
                        PBUF_CFG_INIT(mem_addr, size, dcache_align, use_handshake);     \
        static struct pbuf name = {                                                     \
                .cfg = &cfg_##name,                                                     \
        }

int pbuf_tx_init(struct pbuf *pb);

int pbuf_rx_init(struct pbuf *pb);


int pbuf_write(struct pbuf *pb, const char *buf, uint16_t len);

int pbuf_read(struct pbuf *pb, char *buf, uint16_t len);

uint32_t pbuf_handshake_read(struct pbuf *pb);

void pbuf_handshake_write(struct pbuf *pb, uint32_t value);

int pbuf_get_initial_buf(struct pbuf *pb, volatile char **buf, uint16_t *len);

void pbuf_native_addr_remap(struct pbuf *pb);


#ifdef __cplusplus
}
#endif

#endif /* ZEPHYR_INCLUDE_IPC_PBUF_H_ */
```

## 149.2\_可见注释译文

`_PBUF_IDX_SIZE`表示为区分缓冲区为空和已满而保留不用的数据量。`_PBUF_MIN_DATA_LEN`表示存储最小数据包所需的数据区最小长度：`+1`保证至少有1字节数据，`+_PBUF_IDX_SIZE`用于区分空与满；结果向上取整，使读／写索引始终指向对齐地址。

原生模拟开发板（Native Simulated Board）需要在初始化时修改部分指针，因此`CONFIG_ARCH_POSIX`分支将`PBUF_MAYBE_CONST`定义为空。

`pbuf_cfg`各字段的可见注释如下：

| 参数 | 含义 |
| --- | --- |
| `rd_idx_loc` | 保存`data[]`中第一个有效字节索引的变量地址 |
| `handshake_loc` | 保存握手信息（Handshake Information）的变量地址 |
| `wr_idx_loc` | 保存`data[]`中第一个空闲字节索引的变量地址 |
| `dcache_alignment` | 中央处理器（Central Processing Unit，CPU）数据缓存行大小，单位为字节；用于校验。原注释的待办事项（TODO）：以后用标志替代 |
| `len` | `data[]`长度，单位为字节 |
| `data_loc` | `data[]`所在位置 |

`pbuf_data`保存读／写索引：

| 参数 | 含义 |
| --- | --- |
| `wr_idx` | 用于写入的索引，指向`data[]`中第一个空闲字节 |
| `rd_idx` | 用于读取的索引，指向`data[]`中第一个有效字节 |

`pbuf.cfg`是缓冲区配置；`pbuf.data`是读写缓冲区所用的数据。配置宏中的五条断言英文报错译文见[核间数据包缓冲区接口](P144_核间数据包缓冲区接口.md)；公式、条件编译（Conditional Compilation）、类型限定符（Type Qualifier）和断言原样保留。

`__cplusplus`分支提供 C++ 的 C 语言链接约定；文件末尾注释标记头文件保护宏`ZEPHYR_INCLUDE_IPC_PBUF_H_`的结束。网页没有显示的接口文档注释不补造为源码；其对应说明见接口译文。
