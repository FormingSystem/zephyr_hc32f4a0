# Windows UCRT64；任意目录；P002 完整系统更新已完成。
# 仅原生 Git 缺失或当前窗口选错路径时执行；已有正确选择则跳过。
if ! test -x /ucrt64/bin/git.exe; then
  pacman -S --needed mingw-w64-ucrt-x86_64-git
fi
if test -x /ucrt64/bin/git.exe; then
  export PATH="/ucrt64/bin:$PATH"
  # 已激活 Windows venv 时，保留其 Python 在原生工具目录之前。
  if test -n "${VIRTUAL_ENV:-}"; then
    export PATH="$(cygpath -u "$VIRTUAL_ENV")/Scripts:$PATH"
  fi
  hash -r
else
  printf '原生 Git 尚未可用，停止模块下载并检查安装错误。\n' >&2
fi
type -a git
command -v git
