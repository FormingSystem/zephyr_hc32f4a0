# P014 14.5.5：在已核验 HAL 后检查额外模块发现
# 起点为当前终端 ZEPHYR_BASE 指向的 Zephyr 源码根。
# Windows UCRT64；仅用于已完成 P014 14.3 HC32 HAL 准备的分支。
cd "$ZEPHYR_BASE"
source .venv/Scripts/activate
hal_module_root=../hal_xhsc
if [ ! -f "$hal_module_root/zephyr/module.yml" ]; then
  hal_module_root=../hal_xhsc-zephyr
fi
mkdir -p build/learning-tools/p002
test -f "$hal_module_root/zephyr/module.yml" &&
python scripts/zephyr_module.py --zephyr-base . \
  --extra-modules "$PWD/$hal_module_root" \
  --cmake-out build/learning-tools/p002/zephyr_modules_hc32.txt &&
cat build/learning-tools/p002/zephyr_modules_hc32.txt
